hostname ISP

no shut
exit
int loopback 0
ip add 8.8.8.8 255.0.0.0
exit

! GW Configuration
en
conf t
hostname GW
int e0/0
ip add 200.1.1.2 255.255.255.0
no shut
exit
int e0/1
ip add 192.168.1.1 255.255.255.0
no shut
exit
ip route 0.0.0.0 0.0.0.0 200.1.1.1

🔴NAT Configuration :
-----------------------
! GW NAT CONFIG
! 192.168.1.10 --> 200.1.1.3
! 192.168.1.11 --> 200.1.1.4
! 192.168.1.12 --> 200.1.1.5

ip nat inside source static 192.168.1.10 200.1.1.3
ip nat inside source static 192.168.1.11 200.1.1.4
ip nat inside source static 192.168.1.12 200.1.1.5

int e0/1
ip nat inside

int e0/0
ip nat outside
