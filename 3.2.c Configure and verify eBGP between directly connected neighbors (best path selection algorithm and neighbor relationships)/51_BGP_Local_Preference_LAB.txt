--> with route map we will speciefied the target route that we chose for each Router.
!R3 route map config
en
conf t
router bgp 2
no bgp default local-preference 600
exit

access-list 1 permit 1.1.1.0 0.0.0.255
route-map Lab51 permit 10
match ip add 1
set local-preference 200
exit
route-map Lab51 permit 20
set local-preference 100
exit

router bgp 2
neighbor 192.168.103.1 route-map Lab51 in
do clear ip bgp * soft
