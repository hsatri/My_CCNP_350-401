🔴Basic configurtion:
----------------------
! R1 Config
enable
configure terminal
hostname R1
int e0/0
ip add 192.168.102.1 255.255.255.0
no shut
int e0/1
ip add 192.168.103.1 255.255.255.0
no shut
exit

--------------------------------------------

! R2 Config
enable
configure terminal
hostname R2
int e0/0
ip add 192.168.102.2 255.255.255.0
no shut
exit
int lo 0
ip add 1.1.1.1 255.255.255.0
exit

--------------------------------------------

! R3 Config
enable
configure terminal
hostname R3
int e0/0
ip add 192.168.103.2 255.255.255.0
no shut
exit
int lo 0
ip add 1.1.1.1 255.255.255.0
exit


🔴BGP configurtion:
###################
! R1 BGP config
router bgp 1
neighbor 192.168.102.2 remote-as 2
neighbor 192.168.103.2 remote-as 2

-----------------------------------------

! R2 BGP config
router bgp 2
neighbor 192.168.102.1 remote-as 1
network 1.1.1.0 mask 255.255.255.0


-----------------------------------------

! R3 BGP config
router bgp 2
neighbor 192.168.103.1 remote-as 1
network 1.1.1.0 mask 255.255.255.0

! R3 Distribution config
no network 1.1.1.0 mask 255.255.255.0
distribute connected
