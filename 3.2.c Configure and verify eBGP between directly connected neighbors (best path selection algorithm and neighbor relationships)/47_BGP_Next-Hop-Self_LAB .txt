🔴Basic configurtion:
----------------------
! R1 Config
enable
configure terminal
hostname R1
int e0/0
ip add 192.168.200.1 255.255.255.0
no shut
int e0/1
ip add 192.168.100.1 255.255.255.0
no shut
exit


! R2 Config
enable
configure terminal
hostname R2
int e0/0
ip add 192.168.100.2 255.255.255.0
no shut
exit


! R3 Config
enable
configure terminal
hostname R3
int e0/0
ip add 192.168.200.2 255.255.255.0
no shut
int loopback 0
ip add 3.3.3.3 255.0.0.0
no shut
exit


🔴BGP configurtion:
--------------------
! R1 Configuration
router bgp 1
neighbor 192.168.100.2 remote-as 1
neighbor 192.168.200.2 remote-as 2

! R2 Configuration
router bgp 1
neighbor 192.168.200.1 remote-as 2

! R3 Configuration
router bgp 2
neighbor 192.168.200.1 remote-as 2

🔴BGP Next-Hop-Self configurtion :
----------------------------------
! R1 configurtion
router bgp 1
neighbor 192.168.200.2 next-hop-self 
network 192.168.200.0 mask 255.255.255.0
