🔴Basic Config : 
----------------
! R1 Configuration
en
conf t
hostname R1
interface e0
ip add 192.168.88.1 255.255.255.0
no shut
int lo 0
ip add 1.1.1.1 255.255.255.255

! R2 Configuration
en
conf t
hostname R2
interface e0
ip add 192.168.88.2 255.255.255.0
no shut
int lo 0
ip add 2.2.2.2 255.255.255.255

🔴Eigrp Config : 
----------------
! R1 EIGRP Configuration
en
conf t
router eigrp 1
network 192.168.88.0
network 1.1.1.1
no auto-summary


! R2 EIGRP Configuration
en
conf t
router eigrp 1
network 192.168.88.0
network 2.2.2.2
no auto-summary
