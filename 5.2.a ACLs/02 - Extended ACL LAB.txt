en
conf t

access-list 100 deny tcp host 192.168.1.2 host 172.16.2.2 eq 443
access-list 100 deny tcp host 192.168.1.3 host 172.16.2.3 eq 80 
access-list 100 deny tcp host 192.168.1.3 host 172.16.2.3 eq 443
access-list 100 permit ip any any

int g0/1
